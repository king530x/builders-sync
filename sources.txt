# Builders Sync

Application Android qui synchronise le planning Builders Ingénieurs (connect.builders-ingenieurs.fr)
vers l'agenda Google du téléphone, toutes les 6 heures, sur 30 jours.

L'APK est fabriqué automatiquement par GitHub Actions : voir l'onglet **Releases**.

// Extraction des séances : semaine courante + WEEKS semaines suivantes.
(function (WEEKS) {
  window.__bsResult = '';
  (async function () {
    try {
      const fc = PrimeFaces.widgets.schedule.jqc;
      const all = {};
      const w = ms => new Promise(r => setTimeout(r, ms));
      const idle = async () => {
        await w(1500);
        for (let n = 0; n < 30; n++) {
          const q = PrimeFaces.ajax && PrimeFaces.ajax.Queue;
          if (!q || q.isEmpty()) break;
          await w(500);
        }
        await w(1000);
      };
      const grab = () => fc.fullCalendar('clientEvents').forEach(e => {
        const t = (e.title || '').split('\n');
        if (t.length > 5 && e.start && e.end && t[5].trim())
          all[t[5].trim()] = { id: t[5].trim(), start: e.start.format('YYYY-MM-DD[T]HH:mm:ss'),
                               end: e.end.format('YYYY-MM-DD[T]HH:mm:ss'), lines: t };
      });
      fc.fullCalendar('today'); await idle(); grab();
      for (let i = 0; i < WEEKS; i++) { fc.fullCalendar('next'); await idle(); grab(); }
      fc.fullCalendar('today');
      window.__bsResult = JSON.stringify(Object.values(all));
    } catch (e) {
      window.__bsResult = 'ERR:' + e;
    }
  })();
  return 'started';
})

package fr.dadouch.builderssync;

import android.accounts.Account;
import android.content.ContentResolver;
import android.content.ContentUris;
import android.content.ContentValues;
import android.content.Context;
import android.database.Cursor;
import android.net.Uri;
import android.os.Bundle;
import android.provider.CalendarContract;
import android.provider.CalendarContract.Calendars;
import android.provider.CalendarContract.Events;
import android.provider.CalendarContract.Reminders;

import org.json.JSONObject;

import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

/** Compare le planning avec l'agenda du téléphone et applique uniquement les différences. */
public class CalendarSync {
    private static final Pattern SEANCE = Pattern.compile("séance\\s+(\\d+)");
    private static final int REMINDER_MIN = 30; // rappel par défaut de l'agenda Google

    public static class Cal {
        public long id;
        public String name, account, type;
        public boolean primary;
        public String toString() { return name + (name.equals(account) ? "" : " (" + account + ")"); }
    }

    static class Ev {
        long id, start, end;
        String title, location, desc;
    }

    public static class Result {
        public final List<String> changes = new ArrayList<>();
        public final List<String> warnings = new ArrayList<>();
    }

    /** Agendas modifiables du téléphone. */
    public static List<Cal> calendars(Context c) {
        List<Cal> out = new ArrayList<>();
        String[] proj = {Calendars._ID, Calendars.CALENDAR_DISPLAY_NAME, Calendars.ACCOUNT_NAME,
                Calendars.ACCOUNT_TYPE, Calendars.IS_PRIMARY, Calendars.CALENDAR_ACCESS_LEVEL};
        try (Cursor cur = c.getContentResolver().query(Calendars.CONTENT_URI, proj,
                Calendars.CALENDAR_ACCESS_LEVEL + ">=?",
                new String[]{String.valueOf(Calendars.CAL_ACCESS_CONTRIBUTOR)}, null)) {
            if (cur == null) return out;
            while (cur.moveToNext()) {
                Cal k = new Cal();
                k.id = cur.getLong(0);
                k.name = nz(cur.getString(1));
                k.account = nz(cur.getString(2));
                k.type = nz(cur.getString(3));
                k.primary = cur.getInt(4) == 1;
                out.add(k);
            }
        }
        return out;
    }

    /** Agenda par défaut : l'agenda principal Google (dadouch530@gmail.com de préférence). */
    public static Cal defaultCalendar(Context c) {
        Cal best = null;
        for (Cal k : calendars(c)) {
            if (k.name.equalsIgnoreCase("dadouch530@gmail.com") && "com.google".equals(k.type)) return k;
            if (best == null && k.primary && "com.google".equals(k.type)) best = k;
        }
        return best;
    }

    public static Cal byId(Context c, long id) {
        for (Cal k : calendars(c)) if (k.id == id) return k;
        return null;
    }

    private static String nz(String s) { return s == null ? "" : s; }

    private static Map<String, Ev> existing(Context c, long calId, long from, long to, List<Ev> dupes) {
        Map<String, Ev> map = new HashMap<>();
        String[] proj = {Events._ID, Events.DTSTART, Events.DTEND, Events.TITLE, Events.EVENT_LOCATION, Events.DESCRIPTION};
        String sel = Events.CALENDAR_ID + "=? AND " + Events.DTSTART + ">=? AND " + Events.DTSTART + "<? AND "
                + Events.DELETED + "=0 AND " + Events.DESCRIPTION + " LIKE ?";
        String[] args = {String.valueOf(calId), String.valueOf(from), String.valueOf(to), "%" + Session.TAG + "%"};
        try (Cursor cur = c.getContentResolver().query(Events.CONTENT_URI, proj, sel, args, Events.DTSTART)) {
            if (cur == null) return map;
            while (cur.moveToNext()) {
                Ev e = new Ev();
                e.id = cur.getLong(0);
                e.start = cur.getLong(1);
                e.end = cur.isNull(2) ? e.start : cur.getLong(2);
                e.title = nz(cur.getString(3));
                e.location = nz(cur.getString(4));
                e.desc = nz(cur.getString(5));
                if (!e.desc.contains(Session.TAG)) continue;
                Matcher m = SEANCE.matcher(e.desc);
                if (!m.find()) continue;
                String sid = m.group(1);
                if (map.containsKey(sid)) dupes.add(e); else map.put(sid, e);
            }
        }
        return map;
    }

    public static Result sync(Context c, Cal cal, List<Session> portal, long from, long to) {
        Result r = new Result();
        ContentResolver cr = c.getContentResolver();
        long now = System.currentTimeMillis();
        List<Ev> dupes = new ArrayList<>();
        Map<String, Ev> ex = existing(c, cal.id, from, to, dupes);
        JSONObject snap = Store.snapshot(c);
        JSONObject newSnap = new JSONObject();
        Map<String, Session> byId = new HashMap<>();
        for (Session s : portal) byId.put(s.id, s);

        for (Session s : portal) {
            try { newSnap.put(s.id, new JSONObject().put("t", s.summary()).put("d", s.description())); }
            catch (Exception ignored) { }
            Ev e = ex.get(s.id);
            if (e == null) {
                insert(cr, cal.id, s);
                r.changes.add("Ajout : " + s.label());
                continue;
            }
            if (s.endMs < now) continue; // séance passée : on ne touche pas
            ContentValues cv = new ContentValues();
            List<String> what = new ArrayList<>();
            if (e.start != s.startMs || e.end != s.endMs) {
                cv.put(Events.DTSTART, s.startMs);
                cv.put(Events.DTEND, s.endMs);
                cv.put(Events.EVENT_TIMEZONE, Session.PARIS.getId());
                what.add(Session.day(e.start) + " " + Session.hm(e.start) + "–" + Session.hm(e.end) + " → "
                        + Session.day(s.startMs) + " " + Session.hm(s.startMs) + "–" + Session.hm(s.endMs));
            }
            if (!e.location.trim().equals(s.room)) {
                cv.put(Events.EVENT_LOCATION, s.room);
                what.add("salle " + (e.location.isEmpty() ? "—" : e.location) + " → " + (s.room.isEmpty() ? "—" : s.room));
            }
            JSONObject old = snap.optJSONObject(s.id);
            // Titre / intervenants : comparés au dernier relevé du portail, pour ne pas
            // écraser une correction faite à la main dans l'agenda.
            if (old != null && !old.optString("t").equals(s.summary())) {
                cv.put(Events.TITLE, s.summary());
                what.add("titre « " + old.optString("t") + " » → « " + s.summary() + " »");
            }
            if (old != null && !old.optString("d").equals(s.description())) {
                cv.put(Events.DESCRIPTION, s.description());
                what.add("intervenant/type mis à jour (" + s.intervenants + ")");
            }
            if (cv.size() > 0) {
                cr.update(ContentUris.withAppendedId(Events.CONTENT_URI, e.id), cv, null, null);
                r.changes.add(Session.day(s.startMs) + " " + s.summary() + " : " + String.join(", ", what));
            }
        }

        // Séances disparues du planning = annulées (seulement si elles ne sont pas commencées).
        if (portal.isEmpty() && !ex.isEmpty()) {
            r.warnings.add("Le planning est vide sur la période : aucune suppression effectuée par sécurité.");
        } else {
            for (Map.Entry<String, Ev> en : ex.entrySet()) {
                if (byId.containsKey(en.getKey())) continue;
                Ev e = en.getValue();
                if (e.start <= now) continue;
                cr.delete(ContentUris.withAppendedId(Events.CONTENT_URI, e.id), null, null);
                r.changes.add("Annulé : " + Session.day(e.start) + " " + Session.hm(e.start) + "–" + Session.hm(e.end)
                        + " " + e.title + (e.location.isEmpty() ? "" : " (" + e.location + ")"));
            }
        }
        if (!dupes.isEmpty()) r.warnings.add(dupes.size() + " doublon(s) « Planning Builders » repéré(s), laissés tels quels.");

        // Conserver l'instantané des séances hors période déjà connues.
        java.util.Iterator<String> it = snap.keys();
        while (it.hasNext()) {
            String k = it.next();
            if (!newSnap.has(k) && !ex.containsKey(k)) {
                try { newSnap.put(k, snap.get(k)); } catch (Exception ignored) { }
            }
        }
        Store.saveSnapshot(c, newSnap);

        if (!r.changes.isEmpty()) requestSync(cal);
        return r;
    }

    private static void insert(ContentResolver cr, long calId, Session s) {
        ContentValues cv = new ContentValues();
        cv.put(Events.CALENDAR_ID, calId);
        cv.put(Events.TITLE, s.summary());
        cv.put(Events.DESCRIPTION, s.description());
        cv.put(Events.EVENT_LOCATION, s.room);
        cv.put(Events.DTSTART, s.startMs);
        cv.put(Events.DTEND, s.endMs);
        cv.put(Events.EVENT_TIMEZONE, Session.PARIS.getId());
        cv.put(Events.HAS_ALARM, 1);
        Uri u = cr.insert(Events.CONTENT_URI, cv);
        if (u == null) return;
        ContentValues rv = new ContentValues();
        rv.put(Reminders.EVENT_ID, ContentUris.parseId(u));
        rv.put(Reminders.MINUTES, REMINDER_MIN);
        rv.put(Reminders.METHOD, Reminders.METHOD_ALERT);
        cr.insert(Reminders.CONTENT_URI, rv);
    }

    /** Demande à Android d'envoyer tout de suite les modifications vers Google Agenda. */
    public static void requestSync(Cal cal) {
        try {
            Bundle b = new Bundle();
            b.putBoolean(ContentResolver.SYNC_EXTRAS_MANUAL, true);
            b.putBoolean(ContentResolver.SYNC_EXTRAS_EXPEDITED, true);
            ContentResolver.requestSync(new Account(cal.account, cal.type), CalendarContract.AUTHORITY, b);
        } catch (Exception ignored) { }
    }
}

/** Connexion visible : l'app remplit les identifiants, tu valides Authenticator si demandé. */
public class LoginActivity extends Activity {
    private PortalSession ps;
    private WebView wv;

    @Override
    protected void onCreate(Bundle saved) {
        super.onCreate(saved);
        LinearLayout root = new LinearLayout(this);
        root.setOrientation(LinearLayout.VERTICAL);
        root.setFitsSystemWindows(true);
        TextView status = new TextView(this);
        int p = (int) TypedValue.applyDimension(TypedValue.COMPLEX_UNIT_DIP, 12, getResources().getDisplayMetrics());
        status.setPadding(p, p, p, p);
        status.setTextSize(15);
        status.setText("Connexion…");
        wv = new WebView(this);
        root.addView(status, new LinearLayout.LayoutParams(LinearLayout.LayoutParams.MATCH_PARENT,
                LinearLayout.LayoutParams.WRAP_CONTENT));
        root.addView(wv, new LinearLayout.LayoutParams(LinearLayout.LayoutParams.MATCH_PARENT, 0, 1f));
        setContentView(root);

        ps = new PortalSession(this, wv, false, 15 * 60_000L, 0, new PortalSession.Listener() {
            public void onStatus(String s) { status.setText(s); }
            public void onMfa(String n) {
                status.setText(n.isEmpty() ? "Approuve la connexion dans Microsoft Authenticator."
                        : "Dans Microsoft Authenticator, saisis le numéro " + n + ".");
            }
            public void onReady() {
                Toast.makeText(LoginActivity.this, "Connecté ! Synchronisation lancée.", Toast.LENGTH_LONG).show();
                Store.clearError(LoginActivity.this);
                Notif.cancelError(LoginActivity.this);
                SyncWorker.runNow(LoginActivity.this);
                finish();
            }
            public void onResult(JSONArray a) { }
            public void onError(String code, String m) { status.setText("⚠ " + m); }
        });
        ps.start();
    }

    @Override
    public void onBackPressed() {
        if (wv != null && wv.canGoBack()) wv.goBack();
        else super.onBackPressed();
    }

    @Override
    protected void onDestroy() {
        if (ps != null) ps.cancel();
        if (wv != null) wv.destroy();
        super.onDestroy();
    }
}

        long now = System.currentTimeMillis();
        Session next = null;
        for (Session s : Store.upcoming(c)) {
            if (s.endMs > now) { next = s; break; }
        }
        RemoteViews v = new RemoteViews(c.getPackageName(), R.layout.widget);

import android.app.Notification;
import android.app.NotificationChannel;
import android.app.NotificationManager;
import android.app.PendingIntent;
import android.content.Context;

public class Notif {
    static final String CH_CHANGES = "changes", CH_ERRORS = "errors";
    static final int ID_CHANGES = 1, ID_ERROR = 2, ID_MFA = 3;

    static void channels(Context c) {
        NotificationManager nm = c.getSystemService(NotificationManager.class);
        nm.createNotificationChannel(new NotificationChannel(CH_CHANGES, "Changements de planning",
                NotificationManager.IMPORTANCE_DEFAULT));
        nm.createNotificationChannel(new NotificationChannel(CH_ERRORS, "Connexion et erreurs",
                NotificationManager.IMPORTANCE_HIGH));
    }

    private static PendingIntent open(Context c, Class<?> cls) {
        Intent i = new Intent(c, cls).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK | Intent.FLAG_ACTIVITY_CLEAR_TOP);
        return PendingIntent.getActivity(c, cls.hashCode(), i,
                PendingIntent.FLAG_IMMUTABLE | PendingIntent.FLAG_UPDATE_CURRENT);
    }

    private static void post(Context c, int id, Notification n) {
        try {
            c.getSystemService(NotificationManager.class).notify(id, n);
        } catch (SecurityException ignored) { } // notifications refusées
    }

    static void changes(Context c, List<String> lines) {
        channels(c);
        Notification.InboxStyle st = new Notification.InboxStyle();
        for (String l : lines) st.addLine(l);
        String title = lines.size() == 1 ? "1 changement de planning" : lines.size() + " changements de planning";
        post(c, ID_CHANGES, new Notification.Builder(c, CH_CHANGES)
                .setSmallIcon(android.R.drawable.ic_menu_my_calendar)
                .setContentTitle(title)
                .setContentText(lines.get(0))
                .setStyle(st)
                .setContentIntent(open(c, MainActivity.class))
                .setAutoCancel(true)
                .build());
    }

    static void error(Context c, String message, boolean openLogin) {
        channels(c);
        post(c, ID_ERROR, new Notification.Builder(c, CH_ERRORS)
                .setSmallIcon(android.R.drawable.stat_notify_error)
                .setContentTitle("Builders Sync : synchro impossible")
                .setContentText(message)
                .setStyle(new Notification.BigTextStyle().bigText(message
                        + (openLogin ? "\nTouche ici pour te reconnecter." : "")))
                .setContentIntent(open(c, openLogin ? LoginActivity.class : MainActivity.class))
                .setAutoCancel(true)
                .build());
    }

    static void mfa(Context c, String number) {
        channels(c);
        String txt = number.isEmpty() ? "Ouvre Microsoft Authenticator et approuve la connexion."
                : "Ouvre Microsoft Authenticator et saisis le numéro " + number + ".";
        post(c, ID_MFA, new Notification.Builder(c, CH_ERRORS)
                .setSmallIcon(android.R.drawable.ic_lock_lock)
                .setContentTitle("Validation Microsoft demandée")
                .setContentText(txt)
                .setStyle(new Notification.BigTextStyle().bigText(txt))
                .setTimeoutAfter(4 * 60_000)
                .setAutoCancel(true)
                .build());
    }

    static void cancelMfa(Context c) {
        c.getSystemService(NotificationManager.class).cancel(ID_MFA);
    }

    static void cancelError(Context c) {
        c.getSystemService(NotificationManager.class).cancel(ID_ERROR);
    }

import android.annotation.SuppressLint;
import android.content.Context;

import android.webkit.WebViewClient;

import org.json.JSONArray;
import org.json.JSONObject;
import org.json.JSONTokener;

import java.io.ByteArrayOutputStream;
import java.io.InputStream;
import java.nio.charset.StandardCharsets;

/**
 * Pilote un WebView (visible ou non) : connexion Microsoft automatique avec les identifiants
 * enregistrés, ouverture de « Mon Planning », puis extraction des séances.
 * À utiliser uniquement depuis le thread principal.
 */
public class PortalSession {
    public static final String URL = "https://connect.builders-ingenieurs.fr/faces/Planning.xhtml";

    public interface Listener {
        void onStatus(String status);
        void onMfa(String number);
        void onReady();
        void onResult(JSONArray sessions);
        void onError(String code, String message);
    }

    private final Context ctx;
    private final WebView wv;
    private final Listener l;
    private final boolean extract;
    private final long timeoutMs;
    private final int weeks;
    private final Handler h = new Handler(Looper.getMainLooper());
    private final String email, pwd, driverJs, extractJs;

    private long startedAt;
    private boolean done = false, extracting = false, mfaSeen = false;
    private long extractStartedAt;
    private int pwdSubmits = 0, emailSubmits = 0;
    private String lastHost = "";

    public PortalSession(Context ctx, WebView wv, boolean extract, long timeoutMs, int weeks, Listener l) {
        this.ctx = ctx.getApplicationContext();
        this.wv = wv;
        this.extract = extract;
        this.timeoutMs = timeoutMs;
        this.weeks = weeks;
        this.l = l;
        this.email = Store.email(ctx);
        this.pwd = Store.password(ctx);
        this.driverJs = asset("driver.js");
        this.extractJs = asset("extract.js");
    }

    private String asset(String name) {
        try (InputStream in = ctx.getAssets().open(name)) {
            ByteArrayOutputStream bo = new ByteArrayOutputStream();
            byte[] buf = new byte[8192];
            int n;
            while ((n = in.read(buf)) > 0) bo.write(buf, 0, n);
            return bo.toString(StandardCharsets.UTF_8.name());
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }

    @SuppressLint("SetJavaScriptEnabled")
    public static void configure(WebView wv) {
        WebSettings s = wv.getSettings();
        s.setJavaScriptEnabled(true);
        s.setDomStorageEnabled(true);
        s.setDatabaseEnabled(true);
        s.setLoadWithOverviewMode(true);
        s.setUseWideViewPort(true);
        CookieManager cm = CookieManager.getInstance();
        cm.setAcceptCookie(true);
        cm.setAcceptThirdPartyCookies(wv, true);
    }

    public void start() {
        configure(wv);
        wv.setWebViewClient(new WebViewClient() {
            @Override
            public boolean shouldOverrideUrlLoading(WebView view, WebResourceRequest req) {
                String scheme = req.getUrl().getScheme();
                // Les liens non web (ex. msauth://, intent://) sont ignorés.
                return scheme == null || !(scheme.equals("https") || scheme.equals("http"));
            }

            @Override
            public void onPageStarted(WebView view, String url, Bitmap favicon) {
                status("Chargement de " + host(url) + "…");
            }

            @Override
            public void onPageFinished(WebView view, String url) {
                CookieManager.getInstance().flush();
            }
        });
        startedAt = System.currentTimeMillis();
        status("Ouverture du portail Builders…");
        wv.loadUrl(URL);
        h.postDelayed(this::tick, 3500);
    }

    public void cancel() {
        done = true;
        h.removeCallbacksAndMessages(null);
    }

    private static String host(String url) {
        try {
            String hh = Uri.parse(url).getHost();
            return hh == null ? "" : hh;
        } catch (Exception e) {
            return "";
        }
    }

    private void status(String s) {
        if (!done) l.onStatus(s);
    }

    private void fail(String code, String msg) {
        if (done) return;
        done = true;
        h.removeCallbacksAndMessages(null);
        CookieManager.getInstance().flush();
        l.onError(code, msg);
    }

    private void next(long ms) {
        if (!done) h.postDelayed(this::tick, ms);
    }

    private static String decode(String v) {
        if (v == null || v.equals("null")) return "";
        try {
            Object o = new JSONTokener(v).nextValue();
            return o instanceof String ? (String) o : String.valueOf(o);
        } catch (Exception e) {
            return v;
        }
    }

    private void tick() {
        if (done) return;
        long now = System.currentTimeMillis();
        if (extracting) {
            if (now - extractStartedAt > 120_000) {
                fail("extract_timeout", "La lecture du planning a pris trop de temps.");
                return;
            }
            wv.evaluateJavascript("window.__bsResult||''", v -> {
                String r = decode(v);
                if (r.isEmpty()) { next(2000); return; }
                if (r.startsWith("ERR:")) { fail("extract_error", "Lecture du planning impossible : " + r.substring(4)); return; }
                try {
                    JSONArray a = new JSONArray(r);
                    done = true;
                    h.removeCallbacksAndMessages(null);
                    CookieManager.getInstance().flush();
                    l.onResult(a);
                } catch (Exception e) {
                    fail("extract_error", "Réponse du planning illisible.");
                }
            });
            return;
        }
        if (now - startedAt > timeoutMs) {
            fail(mfaSeen ? "mfa_timeout" : "timeout",
                    mfaSeen ? "La validation Authenticator n'a pas été faite à temps."
                            : "Connexion bloquée sur " + (lastHost.isEmpty() ? "le portail" : lastHost) + ".");
            return;
        }
        String url = wv.getUrl() == null ? "" : wv.getUrl();
        lastHost = host(url);
        boolean msLogin = lastHost.endsWith("login.microsoftonline.com") || lastHost.endsWith("login.live.com");
        String prefix = msLogin
                ? "var __E=" + JSONObject.quote(email) + ",__P=" + JSONObject.quote(pwd) + ";"
                : "var __E='',__P='';";
        wv.evaluateJavascript(prefix + driverJs, v -> handle(decode(v)));
    }

    private void handle(String json) {
        if (done) return;
        String s = "wait", n = "", m = "";
        try {
            JSONObject o = new JSONObject(json);
            s = o.optString("s", "wait");
            n = o.optString("n", "");
            m = o.optString("m", "");
        } catch (Exception ignored) { }
        switch (s) {
            case "ready":
                if (!extract) {
                    done = true;
                    h.removeCallbacksAndMessages(null);
                    CookieManager.getInstance().flush();
                    l.onReady();
                    return;
                }
                status("Lecture du planning…");
                extracting = true;
                extractStartedAt = System.currentTimeMillis();
                wv.evaluateJavascript(extractJs + "(" + weeks + ")", null);
                next(4000);
                return;
            case "sso":
                status("Connexion Microsoft…");
                break;
            case "menu":
                status("Ouverture de « Mon Planning »…");
                break;
            case "email_sub":
                emailSubmits++;
                status("Saisie de l'e-mail…");
                if (emailSubmits > 3) { fail("bad_credentials", "E-mail refusé par Microsoft."); return; }
                break;
            case "pwd_sub":
                pwdSubmits++;
                status("Saisie du mot de passe…");
                if (pwdSubmits > 3) { fail("bad_credentials", "Mot de passe refusé par Microsoft."); return; }
                break;
            case "bad":
                fail("bad_credentials", "Microsoft : " + m);
                return;
            case "mfa":
                if (!mfaSeen) {
                    mfaSeen = true;
                    // On laisse au moins 3 minutes pour approuver.
                    startedAt = Math.max(startedAt, System.currentTimeMillis() - timeoutMs + 180_000);
                    l.onMfa(n);
                }
                status(n.isEmpty() ? "Approuve la connexion dans Authenticator…"
                        : "Approuve dans Authenticator : numéro " + n);
                break;
            case "need_user":
                if (extract) {
                    fail("need_user", m.equals("code")
                            ? "Microsoft demande un code de vérification."
                            : "Identifiants manquants : ouvre l'app pour les enregistrer.");
                    return;
                }
                status(m.equals("code") ? "Saisis le code demandé par Microsoft." : "Saisis tes identifiants.");
                break;
            case "kmsi":
                status("Connexion presque terminée…");
                break;
            default:
                break;
        }
        next(2000);
    }

    /** Crée un WebView invisible dimensionné (nécessaire au rendu du calendrier). */

        return w;
    }

package fr.dadouch.builderssync;

import org.json.JSONArray;
import org.json.JSONException;
import org.json.JSONObject;

import java.time.Instant;
import java.time.LocalDateTime;
import java.time.ZoneId;
import java.time.ZonedDateTime;
import java.time.format.DateTimeFormatter;
import java.util.Locale;

/** Une séance du planning Builders. */
public class Session {
    public static final ZoneId PARIS = ZoneId.of("Europe/Paris");
    public static final String TAG = "[Planning Builders]";

    public String id = "";
    public long startMs, endMs;
    public String room = "", matiere = "", type = "", intervenants = "", note = "";

    /** Construit une séance à partir d'un objet {id,start,end,lines[]} extrait du portail. */
    public static Session fromPortal(JSONObject o) throws JSONException {
        Session s = new Session();
        JSONArray l = o.getJSONArray("lines");
        s.id = o.optString("id", line(l, 5)).trim();
        s.startMs = parseLocal(o.getString("start"));
        s.endMs = parseLocal(o.getString("end"));
        String l0 = line(l, 0);
        int dash = l0.indexOf(" - ");
        if (dash >= 0) s.room = l0.substring(dash + 3).trim();
        else if (l0.endsWith(" -")) s.room = "";
        s.matiere = line(l, 1).trim();
        s.type = line(l, 2).trim();
        s.intervenants = line(l, 3).trim();
        s.note = line(l, 6).trim();
        return s;
    }

    private static String line(JSONArray a, int i) {
        return i < a.length() ? a.optString(i, "") : "";
    }

    static long parseLocal(String iso) {
        return LocalDateTime.parse(iso).atZone(PARIS).toInstant().toEpochMilli();
    }

    public String summary() {
        return note.isEmpty() ? matiere : matiere + " – " + note;
    }

    public String description() {
        StringBuilder b = new StringBuilder();
        b.append(TAG).append(" séance ").append(id);
        if (!type.isEmpty()) b.append("\n").append(type);
        if (!intervenants.isEmpty()) {
            b.append("\n").append(intervenants.contains(",") ? "Intervenants : " : "Intervenant : ")
                    .append(intervenants);
        }
        return b.toString();
    }

    public static ZonedDateTime z(long ms) {
        return Instant.ofEpochMilli(ms).atZone(PARIS);
    }

    public static String hm(long ms) {
        return z(ms).format(DateTimeFormatter.ofPattern("HH'h'mm"));
    }

    public static String day(long ms) {
        String d = z(ms).format(DateTimeFormatter.ofPattern("EEE dd/MM", Locale.FRENCH)).replace(".", "");
        return d.isEmpty() ? d : Character.toUpperCase(d.charAt(0)) + d.substring(1);
    }

    /** Ex. « Mar 10/11 08h00–09h30 Project Management (S202) ». */
    public String label() {
        return day(startMs) + " " + hm(startMs) + "–" + hm(endMs) + " " + summary()
                + (room.isEmpty() ? "" : " (" + room + ")");
    }

    public JSONObject toJson() {
        JSONObject o = new JSONObject();
        try {
            o.put("id", id).put("s", startMs).put("e", endMs).put("room", room).put("mat", matiere)
                    .put("type", type).put("int", intervenants).put("note", note);
        } catch (JSONException ignored) {
        }
        return o;
    }

    public static Session fromJson(JSONObject o) {
        Session s = new Session();
        s.id = o.optString("id");
        s.startMs = o.optLong("s");
        s.endMs = o.optLong("e");
        s.room = o.optString("room");
        s.matiere = o.optString("mat");
        s.type = o.optString("type");
        s.intervenants = o.optString("int");
        s.note = o.optString("note");
        return s;
    }
}

package fr.dadouch.builderssync;

import android.content.Context;
import android.content.SharedPreferences;

import androidx.security.crypto.EncryptedSharedPreferences;
import androidx.security.crypto.MasterKey;

import org.json.JSONArray;
import org.json.JSONException;
import org.json.JSONObject;

import java.text.SimpleDateFormat;
import java.util.ArrayList;
import java.util.Date;
import java.util.List;
import java.util.Locale;

/** Stockage local : identifiants chiffrés + état de la synchro. */
public class Store {
    public static final int DAYS = 30;
    private static final int HISTORY_MAX = 150;

    public static SharedPreferences prefs(Context c) {
        return c.getApplicationContext().getSharedPreferences("state", Context.MODE_PRIVATE);
    }

    private static SharedPreferences secure(Context c) {
        try {
            MasterKey key = new MasterKey.Builder(c.getApplicationContext())
                    .setKeyScheme(MasterKey.KeyScheme.AES256_GCM).build();
            return EncryptedSharedPreferences.create(c.getApplicationContext(), "creds", key,
                    EncryptedSharedPreferences.PrefKeyEncryptionScheme.AES256_SIV,
                    EncryptedSharedPreferences.PrefValueEncryptionScheme.AES256_GCM);
        } catch (Exception e) {
            throw new RuntimeException("Stockage chiffré indisponible", e);
        }
    }

    public static String email(Context c) { return secure(c).getString("email", ""); }
    public static String password(Context c) { return secure(c).getString("pwd", ""); }
    public static boolean hasCreds(Context c) { return !email(c).isEmpty() && !password(c).isEmpty(); }

    public static void saveCreds(Context c, String email, String pwd) {
        secure(c).edit().putString("email", email.trim()).putString("pwd", pwd).apply();
    }

    public static long calendarId(Context c) { return prefs(c).getLong("calId", -1); }
    public static void setCalendarId(Context c, long id, String name) {
        prefs(c).edit().putLong("calId", id).putString("calName", name).apply();
    }
    public static String calendarName(Context c) { return prefs(c).getString("calName", ""); }

    public static void setStatus(Context c, String status) {
        prefs(c).edit().putString("status", status).putLong("statusAt", System.currentTimeMillis()).apply();
    }
    public static String status(Context c) { return prefs(c).getString("status", "Jamais synchronisé"); }
    public static long statusAt(Context c) { return prefs(c).getLong("statusAt", 0); }

    public static void setLastOk(Context c) { prefs(c).edit().putLong("lastOk", System.currentTimeMillis()).apply(); }
    public static long lastOk(Context c) { return prefs(c).getLong("lastOk", 0); }

    // --- Instantané du portail (pour détecter titre/intervenant modifiés) ---
    public static JSONObject snapshot(Context c) {
        try { return new JSONObject(prefs(c).getString("snapshot", "{}")); }
        catch (JSONException e) { return new JSONObject(); }
    }
    public static void saveSnapshot(Context c, JSONObject o) {
        prefs(c).edit().putString("snapshot", o.toString()).apply();
    }

    // --- Prochains cours ---
    public static void saveUpcoming(Context c, List<Session> list) {
        JSONArray a = new JSONArray();
        for (Session s : list) a.put(s.toJson());
        prefs(c).edit().putString("upcoming", a.toString()).apply();
    }
    public static List<Session> upcoming(Context c) {
        List<Session> out = new ArrayList<>();
        try {
            JSONArray a = new JSONArray(prefs(c).getString("upcoming", "[]"));
            for (int i = 0; i < a.length(); i++) out.add(Session.fromJson(a.getJSONObject(i)));
        } catch (JSONException ignored) { }
        return out;
    }

    // --- Historique ---
    public static void addHistory(Context c, List<String> lines) {
        if (lines.isEmpty()) return;
        String stamp = new SimpleDateFormat("dd/MM HH:mm", Locale.FRENCH).format(new Date());
        JSONArray old;
        try { old = new JSONArray(prefs(c).getString("history", "[]")); }
        catch (JSONException e) { old = new JSONArray(); }
        JSONArray a = new JSONArray();
        for (String l : lines) a.put(stamp + " · " + l);
        for (int i = 0; i < old.length() && a.length() < HISTORY_MAX; i++) a.put(old.optString(i));
        prefs(c).edit().putString("history", a.toString()).apply();
    }
    public static List<String> history(Context c) {
        List<String> out = new ArrayList<>();
        try {
            JSONArray a = new JSONArray(prefs(c).getString("history", "[]"));
            for (int i = 0; i < a.length(); i++) out.add(a.optString(i));
        } catch (JSONException ignored) { }
        return out;
    }

    // --- Anti-spam des notifications d'erreur ---
    public static boolean shouldNotifyError(Context c, String code) {
        SharedPreferences p = prefs(c);
        String last = p.getString("errCode", "");
        long at = p.getLong("errAt", 0);
        boolean notify = !code.equals(last) || System.currentTimeMillis() - at > 12L * 3600_000L;
        if (notify) p.edit().putString("errCode", code).putLong("errAt", System.currentTimeMillis()).apply();
        return notify;
    }
    public static void clearError(Context c) {
        prefs(c).edit().remove("errCode").remove("errAt").apply();
    }
}

package fr.dadouch.builderssync;

import android.Manifest;
import android.content.Context;
import android.content.pm.PackageManager;
import android.os.Handler;
import android.os.Looper;
import android.webkit.WebView;

import androidx.annotation.NonNull;
import androidx.work.Constraints;
import androidx.work.ExistingPeriodicWorkPolicy;
import androidx.work.ExistingWorkPolicy;
import androidx.work.NetworkType;
import androidx.work.OneTimeWorkRequest;
import androidx.work.PeriodicWorkRequest;
import androidx.work.WorkManager;
import androidx.work.Worker;
import androidx.work.WorkerParameters;

import org.json.JSONArray;

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicBoolean;

public class SyncWorker extends Worker {
    private static final AtomicBoolean RUNNING = new AtomicBoolean(false);
    private static final long LOGIN_TIMEOUT = 150_000;

    public SyncWorker(@NonNull Context c, @NonNull WorkerParameters p) { super(c, p); }

    private static Constraints net() {
        return new Constraints.Builder().setRequiredNetworkType(NetworkType.CONNECTED).build();
    }

    public static void schedule(Context c) {
        PeriodicWorkRequest req = new PeriodicWorkRequest.Builder(SyncWorker.class, 6, TimeUnit.HOURS)
                .setConstraints(net()).build();
        WorkManager.getInstance(c).enqueueUniquePeriodicWork("periodic", ExistingPeriodicWorkPolicy.KEEP, req);
    }

    public static void runNow(Context c) {
        OneTimeWorkRequest req = new OneTimeWorkRequest.Builder(SyncWorker.class).setConstraints(net()).build();
        WorkManager.getInstance(c).enqueueUniqueWork("now", ExistingWorkPolicy.KEEP, req);
    }

    public static boolean isRunning() { return RUNNING.get(); }

    @NonNull
    @Override
    public Result doWork() {
        if (!RUNNING.compareAndSet(false, true)) return Result.success();
        try {
            run(getApplicationContext());
        } catch (Exception e) {
            fail(getApplicationContext(), "crash", "Erreur inattendue : " + e.getMessage(), false);
        } finally {
            RUNNING.set(false);
        }
        return Result.success();
    }

    private static void fail(Context c, String code, String msg, boolean login) {
        Store.setStatus(c, "⚠ " + msg);
        Store.addHistory(c, java.util.Collections.singletonList("Erreur : " + msg));
        if (Store.shouldNotifyError(c, code)) Notif.error(c, msg, login);
    }

    private void run(Context c) throws InterruptedException {
        if (c.checkSelfPermission(Manifest.permission.WRITE_CALENDAR) != PackageManager.PERMISSION_GRANTED
                || c.checkSelfPermission(Manifest.permission.READ_CALENDAR) != PackageManager.PERMISSION_GRANTED) {
            fail(c, "perm", "Autorisation d'accès à l'agenda manquante : ouvre l'app.", false);
            return;
        }
        if (!Store.hasCreds(c)) {
            fail(c, "creds", "Identifiants Microsoft non enregistrés : ouvre l'app.", false);
            return;
        }
        CalendarSync.Cal cal = CalendarSync.byId(c, Store.calendarId(c));
        if (cal == null) cal = CalendarSync.defaultCalendar(c);
        if (cal == null) {
            fail(c, "nocal", "Aucun agenda Google trouvé sur le téléphone.", false);
            return;
        }

        Store.setStatus(c, "Synchronisation en cours…");
        final JSONArray[] result = {null};
        final String[] err = {null, null};
        final CountDownLatch latch = new CountDownLatch(1);
        final WebView[] holder = {null};
        final PortalSession[] ps = {null};
        int weeks = Store.DAYS / 7 + 2;
        Handler main = new Handler(Looper.getMainLooper());
        main.post(() -> {
            try {
                holder[0] = PortalSession.headless(c);
                ps[0] = new PortalSession(c, holder[0], true, LOGIN_TIMEOUT, weeks, new PortalSession.Listener() {
                    public void onStatus(String s) { }
                    public void onMfa(String n) { Notif.mfa(c, n); }
                    public void onReady() { }
                    public void onResult(JSONArray a) { result[0] = a; latch.countDown(); }
                    public void onError(String code, String m) { err[0] = code; err[1] = m; latch.countDown(); }
                });
                ps[0].start();
            } catch (Exception e) {
                err[0] = "webview";
                err[1] = "Navigateur interne indisponible : " + e.getMessage();
                latch.countDown();
            }
        });
        boolean finished = latch.await(9, TimeUnit.MINUTES);
        main.post(() -> {
            if (ps[0] != null) ps[0].cancel();
            if (holder[0] != null) { holder[0].stopLoading(); holder[0].destroy(); }
        });
        Notif.cancelMfa(c);
        if (!finished) { fail(c, "timeout", "Délai dépassé pendant la synchro.", false); return; }
        if (err[0] != null) {
            boolean login = err[0].startsWith("mfa") || err[0].equals("timeout") || err[0].equals("need_user")
                    || err[0].equals("bad_credentials");
            fail(c, err[0], err[1] + " Rien n'a été modifié dans l'agenda.", login);
            return;
        }

        // Filtrer la période : aujourd'hui 00:00 → aujourd'hui + DAYS jours inclus.
        LocalDate today = LocalDate.now(Session.PARIS);
        long from = today.atStartOfDay(Session.PARIS).toInstant().toEpochMilli();
        long to = today.plusDays(Store.DAYS + 1).atStartOfDay(Session.PARIS).toInstant().toEpochMilli();
        List<Session> list = new ArrayList<>();
        try {
            for (int i = 0; i < result[0].length(); i++) {
                Session s = Session.fromPortal(result[0].getJSONObject(i));
                if (s.startMs >= from && s.startMs < to && !s.id.isEmpty()) list.add(s);
            }
        } catch (Exception e) {
            fail(c, "format", "Format du planning inattendu (" + e.getMessage() + ").", false);
            return;
        }
        list.sort(Comparator.comparingLong(s -> s.startMs));

        CalendarSync.Result r = CalendarSync.sync(c, cal, list, from, to);
        Store.saveUpcoming(c, list);
        Store.setLastOk(c);
        Store.clearError(c);
        Notif.cancelError(c);
        List<String> hist = new ArrayList<>(r.changes);
        hist.addAll(r.warnings);
        Store.addHistory(c, hist.isEmpty()
                ? java.util.Collections.singletonList("Synchro OK : aucun changement (" + list.size() + " séances)")
                : hist);
        Store.setStatus(c, r.changes.isEmpty()
                ? "À jour : " + list.size() + " séances, aucun changement"
                : "À jour : " + r.changes.size() + " changement(s) appliqué(s)");
        if (!r.changes.isEmpty()) Notif.changes(c, r.changes);
        NextCourseWidget.updateAll(c);
    }
}
